\section{Introduction \& Motivation}
% =============================================================

% ─────────────────────────────────────────────────────────────
% SLIDE 3: Introduction
% ─────────────────────────────────────────────────────────────
\begin{frame}{Introduction \& Project Motivation}
  \begin{itemize}
    \item \textbf{The Modern Fitness Dilemma}: Physical health and gym participation have surged globally; however, over 65\% of beginners abandon gym memberships within three months due to lack of personalized guidance, inaccurate nutrition plans, and disorganized workout routines.
    \item \textbf{Flaws of Existing Gym Systems}:
    \begin{itemize}
      \item Commercial gym software primarily handles billing and turnstile access, completely neglecting scientific member guidance.
      \item Generic static diet charts fail to calculate personalized Basal Metabolic Rate (BMR) and Total Daily Energy Expenditure (TDEE).
      \item Complex multi-page nutrition apps induce cognitive fatigue through endless vertical scrolling and cluttered interfaces.
    \end{itemize}
    \item \textbf{The GymFit Solution}: An integrated, full-stack digital fitness ecosystem uniting scientific nutrition modeling, interactive 3D exercise biomechanics, and robust administrative management into one cohesive web platform.
  \end{itemize}
\end{frame}

% ─────────────────────────────────────────────────────────────
% SLIDE 4: Problem Statement
% ─────────────────────────────────────────────────────────────
\begin{frame}{Problem Statement}
  \begin{block}{Core Engineering Problem}
    To engineer a responsive, high-performance, and scientifically verified full-stack fitness management application that:
  \end{block}
  \begin{enumerate}
    \item \textbf{Eliminates Nutrition Guesswork}: Eliminates arbitrary calorie targets by implementing the gold-standard Mifflin--St Jeor metabolic equations and empirical macronutrient partitioning for gym users.
    \item \textbf{Solves UI Scroll Fatigue}: Delivers a single-screen, zero-scroll 3-box daily diet architecture (Breakfast, Mid-Day, Evening Recovery) with real-time food swapping.
    \item \textbf{Enhances Form \& Technique}: Bridges anatomical exercise knowledge with 3D biomechanical visual demonstrations (e.g., Incline Dumbbell Press).
    \item \textbf{Streamlines Facility Management}: Delivers automated administrative governance across memberships, revenue tracking, and equipment maintenance.
  \end{enumerate}
\end{frame}

\section{Nutritional Science \& Mathematical Modeling}
% =============================================================

% ─────────────────────────────────────────────────────────────
% SLIDE 8: Mifflin-St Jeor BMR Equation
% ─────────────────────────────────────────────────────────────
\begin{frame}{Nutritional Science: Mifflin--St Jeor BMR Formulation}
  \begin{itemize}
    \item The \textbf{Mifflin--St Jeor Equation} is validated as the most accurate clinical formula for estimating Basal Metabolic Rate (BMR) in healthy individuals.
  \end{itemize}
  \begin{block}{Scientific Mathematical Formulations}
    \begin{align*}
      \text{For Males:} \quad & \mathbf{BMR = 10W + 6.25H - 5A + 5} \\
      \text{For Females:} \quad & \mathbf{BMR = 10W + 6.25H - 5A - 161}
    \end{align*}
    \vspace{-0.3cm}
    \begin{itemize}
      \item $W =$ Body Weight in kilograms ($\text{kg}$)
      \item $H =$ Stature/Height in centimeters ($\text{cm}$)
      \item $A =$ Chronological Age in years ($\text{yrs}$)
    \end{itemize}
  \end{block}
  \begin{exampleblock}{Benchmark System Verification}
    For a 24-year-old male athlete weighing $80\text{ kg}$ with height $175\text{ cm}$:
    \[
      \text{BMR} = 10(80) + 6.25(175) - 5(24) + 5 = 800 + 1093.75 - 120 + 5 = \mathbf{1778.75\text{ kcal/day}} \approx \mathbf{1779\text{ kcal}}
    \]
  \end{exampleblock}
\end{frame}

% ─────────────────────────────────────────────────────────────
% SLIDE 9: TDEE Physical Activity Multipliers
% ─────────────────────────────────────────────────────────────
\begin{frame}{Total Daily Energy Expenditure (TDEE)}
  \begin{itemize}
    \item Total Daily Energy Expenditure accounts for non-exercise activity thermogenesis (NEAT) and resistance exercise thermogenesis:
    \[
      \mathbf{TDEE = BMR \times ActivityFactor}
    \]
  \end{itemize}
  \begin{table}[h]
    \centering
    \small
    \begin{tabular}{lcl}
      \toprule
      \textbf{Activity Category} & \textbf{Multiplier} & \textbf{Weekly Training Description} \\
      \midrule
      \textbf{Sedentary} & $\times 1.200$ & Little or no intentional physical exercise \\
      \textbf{Lightly Active} & $\times 1.375$ & Light exercise / gym training 1--3 days per week \\
      \textbf{Moderately Active} & $\mathbf{\times 1.550}$ & Moderate gym workouts 3--5 days per week \\
      \textbf{Very Active} & $\times 1.725$ & Intense resistance training 6--7 days per week \\
      \textbf{Extremely Active} & $\times 1.900$ & Double daily training / highly demanding physical labor \\
      \bottomrule
    \end{tabular}
  \end{table}
  \begin{block}{Goal-Specific Energy Calibration}
    \begin{itemize}
      \item \textbf{Muscle Gain / Bulking}: $\text{Target} = \text{TDEE} + 350\text{ kcal}$ (Controlled caloric surplus)
      \item \textbf{Fat Loss / Cutting}: $\text{Target} = \text{TDEE} - 400\text{ kcal}$ (Deficit preserving lean tissue)
      \item \textbf{Weight Maintenance}: $\text{Target} = \text{TDEE} + 0\text{ kcal}$ (Isocaloric homeostasis)
    \end{itemize}
  \end{block}
\end{frame}

% ─────────────────────────────────────────────────────────────
% SLIDE 10: Gym Macronutrient Partitioning
% ─────────────────────────────────────────────────────────────
\begin{frame}{Gym-Specific Macronutrient Partitioning Model}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Macronutrient Energy Density Constants}:
      \begin{equation*}
        \text{Protein} = 4\text{ kcal/g}, \quad \text{Carbohydrate} = 4\text{ kcal/g}, \quad \text{Fat} = 9\text{ kcal/g}
      \end{equation*}
      \textbf{Three-Step Allocation Protocol}:
      \begin{enumerate}
        \item \textbf{Dietary Protein (1.4--2.0 g/kg)}:
        \[
          P_{\text{grams}} = W_{\text{kg}} \times 2.0, \quad P_{\text{kcal}} = P_{\text{grams}} \times 4
        \]
        \item \textbf{Essential Dietary Fat (20--35\% total calories)}:
        \[
          F_{\text{kcal}} = 0.25 \times \text{Target}_{\text{kcal}}, \quad F_{\text{grams}} = \frac{F_{\text{kcal}}}{9}
        \]
        \item \textbf{Glycogen-Sparing Carbohydrates (Remaining balance)}:
        \[
          C_{\text{kcal}} = \text{Target}_{\text{kcal}} - (P_{\text{kcal}} + F_{\text{kcal}}), \quad C_{\text{grams}} = \frac{C_{\text{kcal}}}{4}
        \]
      \end{enumerate}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{exampleblock}{Verified Test Case Allocation}
        \textbf{Profile}: Male, $80\text{ kg}$, Bulking Target: $3107\text{ kcal}$
        \begin{itemize}
          \item \textbf{Protein}: $80 \times 2.0 = \mathbf{160\text{g}}$ ($640\text{ kcal}$, 21\%)
          \item \textbf{Fat}: $25\% \times 3107 = \mathbf{86\text{g}}$ ($777\text{ kcal}$, 25\%)
          \item \textbf{Carbs}: $(3107 - 640 - 777)/4 = \mathbf{423\text{g}}$ ($1690\text{ kcal}$, 54\%)
        \end{itemize}
      \end{exampleblock}
      \begin{alertblock}{Hydration \& Fiber Standards}
        \begin{itemize}
          \item Hydration: $40\text{ mL/kg bodyweight}$ ($3.2\text{ L/day}$)
          \item Dietary Fiber: $14\text{ g per } 1000\text{ kcal}$ ($43\text{ g/day}$)
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 13: Smart Food Replacement Engine
% ─────────────────────────────────────────────────────────────
\begin{frame}{Smart Food Replacement Engine}
  \begin{itemize}
    \item When users face allergies, food unavailability, or dietary restrictions (e.g., swapping Paneer for Tofu or Soya), the replacement algorithm computes a macro-distance penalty score:
  \end{itemize}
  \begin{block}{Macro-Distance Optimization Function}
    \[
      \mathbf{\text{Score}} = \alpha \cdot |\text{Cal}_{\text{candidate}} - \text{Cal}_{\text{current}}| + \beta \cdot |\text{Prot}_{\text{candidate}} - \text{Prot}_{\text{current}}|
    \]
    Where $\alpha = 1.0$ and $\beta = 4.0$ (assigning higher weight to protein preservation for gym athletes).
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Automated Constraint Filtering}:
      \begin{itemize}
        \item Category matching (Dairy, Grain, Poultry, Vegan).
        \item Exclusion of declared user allergens (e.g., Lactose, Gluten, Nuts).
        \item Automatic recalculation of meal and daily totals upon substitution.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Verified Nutrition Database}:
      \begin{itemize}
        \item Over 50 culturally tailored food items (Indian and Western fitness foods).
        \item Explicit raw vs. cooked weight measurements.
        \item Complete macro breakdowns (Calories, Protein, Carbs, Fat, Fiber).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
